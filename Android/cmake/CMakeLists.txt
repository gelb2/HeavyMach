cmake_minimum_required(VERSION 3.20)
project(HeavyMachAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Debug APKs retain symbols and run-as, while gameplay uses optimized code.
add_compile_options(-O2)
get_filename_component(HM_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
option(HM_ANDROID_VALIDATION "Build separate cheat application" OFF)
set(HM_BUILD_VALIDATION_TESTS ${HM_ANDROID_VALIDATION} CACHE BOOL "" FORCE)
add_subdirectory("${HM_ROOT}" core EXCLUDE_FROM_ALL)
set(LIBTYPE STATIC CACHE STRING "" FORCE)
foreach(flag ALSOFT_UTILS ALSOFT_EXAMPLES ALSOFT_INSTALL ALSOFT_BACKEND_WAVE)
 set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(ALSOFT_NO_CONFIG_UTIL ON CACHE BOOL "" FORCE)
set(ALSOFT_REQUIRE_OPENSL ON CACHE BOOL "" FORCE)
add_subdirectory("${HM_ROOT}/vendor/openal-soft" openal EXCLUDE_FROM_ALL)
set(NATIVE "${CMAKE_CURRENT_LIST_DIR}/../app/src/main/cpp")
add_library(heavymach SHARED ${NATIVE}/Bridge.cpp ${NATIVE}/Runtime.cpp ${NATIVE}/Media.cpp ${NATIVE}/Audio.cpp)
target_include_directories(heavymach PRIVATE "${HM_ROOT}/Sources/Core" "${HM_ROOT}/Sources/iOS" "${HM_ROOT}/vendor/stb" "${HM_ROOT}/vendor/openal-soft/include")
target_compile_options(heavymach PRIVATE -ffp-contract=off -fno-fast-math -Wall -Wextra)
target_link_options(heavymach PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
if(HM_ANDROID_VALIDATION)
 target_compile_definitions(heavymach PRIVATE HM_VALIDATION=1)
 target_link_libraries(heavymach PRIVATE HeavyMachValidationCore)
else()
 target_compile_definitions(heavymach PRIVATE HM_VALIDATION=0)
 target_link_libraries(heavymach PRIVATE HeavyMachCore)
endif()
target_link_libraries(heavymach PRIVATE OpenAL EGL GLESv1_CM android log)

add_subdirectory("${HM_ROOT}/Modules/SaveTransfer" transfer)
target_link_libraries(heavymach PRIVATE HeavyMachSaveTransfer)
